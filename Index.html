Langkah Penghubung Google Spreadsheet

Agar data langsung masuk ke link Google Sheet Anda (131_7jVY0ZEe0pM9G1cZNZiGhPWds_UFZ_zWBXh_JpVM), lakukan 2 langkah mudah berikut:

Langkah 1: Pasang Skrip di Google Sheet Anda

1. Buka link Google Sheet Anda, lalu klik menu atas: Extensions (Ekstensi) > Apps Script.

2. Hapus seluruh kode bawaan, lalu salin dan tempel (paste) kode lengkap berikut:

function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); try { var data = JSON.parse(e.postData.contents); if (sheet.getLastRow() === 0) { sheet.appendRow([ "Timestamp", "Nama Mahasiswa", "NIM", "Prodi / Sekolah", "Aspek 1", "Aspek 2", "Aspek 3", "Aspek 4", "Aspek 5", "Aspek 6", "Aspek 7", "Aspek 8", "Aspek 9", "Aspek 10", "Total Skor", "Catatan Komentar" ]); } sheet.appendRow([ data.timestamp || new Date(), data.namaMahasiswa, data.nimMahasiswa, data.prodiSekolah, data.Aspek_1, data.Aspek_2, data.Aspek_3, data.Aspek_4, data.Aspek_5, data.Aspek_6, data.Aspek_7, data.Aspek_8, data.Aspek_9, data.Aspek_10, data.totalSkor, data.catatanKomentar ]); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService.createTextOutput("Web App Apps Script aktif dan siap menerima data POST dari Form Penilaian UNY.") .setMimeType(ContentService.MimeType.TEXT); }

Langkah 2: Lakukan New Deployment (PENTING!)

3. Di pojok kanan atas Apps Script, klik tombol Deploy > New deployment (Deployment baru).

4. Pilih jenis Web app. Pada Execute as pilih Me (Email Anda) dan Who has access pilih Anyone (Siapa saja).

5. Klik Deploy / Update, lalu salin URL Web App terbaru berformat https://script.google.com/macros/s/.../exec dan tempel di bawah ini.
